   178→/**
   179→ * ServiceBasket — floating basket button + RFQ modal.
   180→ * `accent` rebrands the chrome: default #FF4500 (alpha/detail pages),
   181→ * beta mode mounts it with its gold #D4AF37 accent.
   182→ */
   183→export function ServiceBasket({ accent = "#FF4500" }: { accent?: string }) {
   184→  const { items, isOpen, toggleOpen, removeItem, clearBasket, getTotal, getServicesTotal, getAddonsTotal, getDiscount, getDiscountedTotal, getServiceCount, getAddonCount, haggleUsed, haggleRoll, haggleDiscountRate, getHaggleDiscount, getGrandTotal } = useBasketStore();
   185→  const [submitting, setSubmitting] = React.useState(false);
   186→  const [btnHover, setBtnHover] = React.useState(false);
   187→
   188→  // === A11Y: focus management for the basket modal ===
   189→  // - toggleBtnRef:  the floating basket button — focus restored here on close.
   190→  // - closeBtnRef:    the modal's close button — autofocus on open so keyboard
   191→  //                   users land inside the dialog immediately.
   192→  // - panelRef:       the modal panel root — used for the Tab focus trap.
   193→  // - lastFocusedRef: the element that had focus before the modal opened
   194→  //                   (usually the floating basket button or a service card's
   195→  //                   "ADD TO QUOTE" button). Restored on close.
   196→  const toggleBtnRef = React.useRef<HTMLButtonElement | null>(null);
   197→  const closeBtnRef = React.useRef<HTMLButtonElement | null>(null);
   198→  const panelRef = React.useRef<HTMLDivElement | null>(null);
   199→  const lastFocusedRef = React.useRef<HTMLElement | null>(null);
   200→
   201→  // Track the previously-focused element + autofocus the close button on open.
   202→  React.useEffect(() => {
   203→    if (!isOpen) return;
   204→    lastFocusedRef.current = (document.activeElement as HTMLElement) ?? null;
   205→    // Defer one tick so the modal panel has rendered.
   206→    const t = setTimeout(() => {
   207→      closeBtnRef.current?.focus();
   208→    }, 0);
   209→    return () => clearTimeout(t);
   210→  }, [isOpen]);
   211→
   212→  // Restore focus to the trigger element on close (WCAG 2.4.3 + standard a11y).
   213→  React.useEffect(() => {
   214→    if (isOpen) return;
   215→    if (lastFocusedRef.current && typeof lastFocusedRef.current.focus === "function") {
   216→      lastFocusedRef.current.focus();
   217→      lastFocusedRef.current = null;
   218→    }
   219→  }, [isOpen]);
   220→
   221→  // Tab focus trap + Escape-to-close — active only while the modal is open.
   222→  React.useEffect(() => {
   223→    if (!isOpen) return;
   224→    const onKey = (e: KeyboardEvent) => {
   225→      if (e.key === "Escape") {
   226→        e.preventDefault();
   227→        toggleOpen();
   228→        return;
   229→      }
   230→      if (e.key !== "Tab") return;
   231→      const panel = panelRef.current;
   232→      if (!panel) return;
   233→      // Query visible + non-disabled focusables inside the panel.
   234→      const focusables = Array.from(
   235→        panel.querySelectorAll<HTMLElement>(
   236→          'a[href], button:not([disabled]), textarea:not([disabled]), input:not([disabled]), select:not([disabled]), [tabindex]:not([tabindex="-1"])'
   237→        )
   238→      ).filter((el) => el.offsetParent !== null || el === document.activeElement);
   239→      if (focusables.length === 0) return;
   240→      const first = focusables[0];
   241→      const last = focusables[focusables.length - 1];
   242→      const active = document.activeElement as HTMLElement | null;
   243→      if (e.shiftKey) {
   244→        if (active === first || !panel.contains(active)) {
   245→          e.preventDefault();
   246→          last.focus();
   247→        }
   248→      } else {
   249→        if (active === last) {
   250→          e.preventDefault();
   251→          first.focus();
   252→        }
   253→      }
   254→    };
   255→    window.addEventListener("keydown", onKey);
   256→    return () => window.removeEventListener("keydown", onKey);
   257→  }, [isOpen, toggleOpen]);
   258→
   259→  const submitRFQ = async () => {
   260→    if (items.length === 0) return;
   261→    setSubmitting(true);
   262→    try {
   263→      const res = await fetch("/api/sigma/transmit?XTransformPort=3000", {
   264→        method: "POST",
   265→        headers: { "Content-Type": "application/json" },
   266→        body: JSON.stringify({
   267→          identity: "basket-rfq",
   268→          channel: "RFQ",
   269→          message: JSON.stringify({
   270→            type: "quotation_request",
   271→            items: items.map((i) => ({ slug: i.slug, name: i.name, type: i.type, addonType: i.addonType, price: i.price })),
   272→            servicesTotal: getServicesTotal(),
   273→            addonsTotal: getAddonsTotal(),
   274→            total: getTotal(),
   275→            discount: getDiscount(),
   276→            final: getDiscountedTotal(),
   277→            serviceCount: getServiceCount(),
   278→            addonCount: getAddonCount(),
   279→            haggleUsed,
   280→            haggleRoll,
   281→            haggleDiscountRate,
   282→            haggleDiscount: getHaggleDiscount(),
   283→            grandTotal: getGrandTotal(),
   284→          }),
   285→        }),
   286→      });
   287→      const data = await res.json();
   288→      if (data.ok) {
   289→        toast.success("▮ QUOTATION REQUESTED", { description: `Ref: ${data.ref}` });
   290→        clearBasket();
   291→        toggleOpen();
   292→      }
   293→    } catch {
   294→      toast.error("SUBMISSION FAILED");
   295→    } finally {
   296→      setSubmitting(false);
   297→    }
   298→  };
   299→
   300→  const services = items.filter((i) => i.type === "service");
   301→  const addons = items.filter((i) => i.type === "addon");
   302→
   303→  return (
   304→    <>
   305→      {/* Floating basket button — lg:left-20 clears beta mode's left rail */}
   306→      <button
   307→        ref={toggleBtnRef}
   308→        onClick={toggleOpen}
   309→        onMouseEnter={() => setBtnHover(true)}
