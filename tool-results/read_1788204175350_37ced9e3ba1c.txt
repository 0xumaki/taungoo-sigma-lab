   343→// Per-package add-to-basket button — each of STARTER/PRO/ENTERPRISE can be added individually.
   344→// Adding STARTER of AI Chatbot + ENTERPRISE of Voice AI = 2 main services = 7% discount.
   345→//
   346→// Currency mode behavior:
   347→//   - LOCAL  → adds the LOCAL (MMK) price to the basket. Basket is MMK-only.
   348→//   - INTL   → international orders are not self-serve (USD invoicing, custom onboarding).
   349→//              The button becomes "REQUEST QUOTE" and opens the contact modal.
   350→function PackageAddButton({
   351→  slug,
   352→  name,
   353→  icon,
   354→  price,
   355→  intlPrice,
   356→  currencyMode,
   357→  popular,
   358→  onRequestQuote,
   359→}: {
   360→  slug: string;
   361→  name: string;
   362→  icon: string;
   363→  price: string;
   364→  intlPrice?: string;
   365→  currencyMode: CurrencyMode;
   366→  popular?: boolean;
   367→  onRequestQuote: () => void;
   368→}) {
   369→  const { items, addItem } = useBasketStore();
   370→  const inBasket = items.some((i) => i.slug === slug);
   371→  const priceNum = parsePrice(price);
   372→  const isCustom = price === "custom";
   373→  // In INTL mode, the displayed intlPrice (if it's "custom" or absent) also routes to quote
   374→  const isIntlCustom = currencyMode === "INTL" && (!intlPrice || intlPrice === "custom");
   375→  const routeToQuote = isCustom || isIntlCustom || currencyMode === "INTL";
   376→
   377→  const handleAdd = () => {
   378→    if (routeToQuote) {
   379→      if (currencyMode === "INTL") {
   380→        toast.info("▮ INTERNATIONAL USD — REQUEST A QUOTE");
   381→      } else if (isCustom) {
   382→        toast.info("▮ CUSTOM PRICING — CONTACT OUR TEAM");
   383→      }
   384→      onRequestQuote();
   385→      return;
   386→    }
   387→    addItem({ slug, name, type: "service", price: priceNum, icon });
   388→    toast.success(`▮ ${name} ADDED TO BASKET`);
   389→  };
   390→
   391→  const buttonLabel = inBasket
   392→    ? "IN BASKET"
   393→    : currencyMode === "INTL"
   394→      ? "REQUEST QUOTE"
   395→      : isCustom
   396→        ? "REQUEST QUOTE"
   397→        : "ADD TO BASKET";
   398→
   399→  return (
   400→    <button
   401→      onClick={handleAdd}
   402→      disabled={inBasket}
   403→      className={`mt-4 flex w-full items-center justify-center gap-2 border py-2.5 font-mono text-[10px] uppercase tracking-[0.18em] transition-all ${
   404→        inBasket
   405→          ? "border-[#00FF94] bg-[#00FF94]/10 text-[#00FF94]"
   406→          : currencyMode === "INTL"
   407→            ? "border-foreground/60 text-foreground hover:border-[#FF4500] hover:text-[#FF4500]"
   408→            : popular
   409→              ? "border-[#FF4500] bg-[#FF4500] text-black hover:opacity-80"
   410→              : "border-foreground/60 text-foreground hover:border-[#FF4500] hover:text-[#FF4500]"
   411→      }`}
   412→    >
   413→      {inBasket ? (
   414→        <>
   415→          <Check className="h-3 w-3" /> IN BASKET
   416→        </>
   417→      ) : (
   418→        <>
   419→          <Plus className="h-3 w-3" /> {buttonLabel}
   420→        </>
   421→      )}
   422→    </button>
   423→  );
   424→}
