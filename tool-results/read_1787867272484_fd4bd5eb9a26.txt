   328→   328→//              The button becomes "REQUEST QUOTE" and opens the contact modal.
   329→   329→function PackageAddButton({
   330→   330→  slug,
   331→   331→  name,
   332→   332→  icon,
   333→   333→  price,
   334→   334→  intlPrice,
   335→   335→  currencyMode,
   336→   336→  popular,
   337→   337→  onRequestQuote,
   338→   338→}: {
   339→   339→  slug: string;
   340→   340→  name: string;
   341→   341→  icon: string;
   342→   342→  price: string;
   343→   343→  intlPrice?: string;
   344→   344→  currencyMode: CurrencyMode;
   345→   345→  popular?: boolean;
   346→   346→  onRequestQuote: () => void;
   347→   347→}) {
   348→   348→  const { items, addItem } = useBasketStore();
   349→   349→  const inBasket = items.some((i) => i.slug === slug);
   350→   350→  const priceNum = parsePrice(price);
   351→   351→  const isCustom = price === "custom";
   352→   352→  // In INTL mode, the displayed intlPrice (if it's "custom" or absent) also routes to quote
   353→   353→  const isIntlCustom = currencyMode === "INTL" && (!intlPrice || intlPrice === "custom");
   354→   354→  const routeToQuote = isCustom || isIntlCustom || currencyMode === "INTL";
   355→   355→
   356→   356→  const handleAdd = () => {
   357→   357→    if (routeToQuote) {
   358→   358→      if (currencyMode === "INTL") {
   359→   359→        toast.info("▮ INTERNATIONAL USD — REQUEST A QUOTE");
   360→   360→      } else if (isCustom) {
   361→   361→        toast.info("▮ CUSTOM PRICING — CONTACT OUR TEAM");
   362→   362→      }
   363→   363→      onRequestQuote();
   364→   364→      return;
   365→   365→    }
   366→   366→    addItem({ slug, name, type: "service", price: priceNum, icon });
   367→   367→    toast.success(`▮ ${name} ADDED TO BASKET`);
   368→   368→  };
   369→   369→
   370→   370→  const buttonLabel = inBasket
   371→   371→    ? "IN BASKET"
   372→   372→    : currencyMode === "INTL"
   373→   373→      ? "REQUEST QUOTE"
   374→   374→      : isCustom
   375→   375→        ? "REQUEST QUOTE"
   376→   376→        : "ADD TO BASKET";
   377→   377→
   378→   378→  return (
   379→   379→    <button
   380→   380→      onClick={handleAdd}
   381→   381→      disabled={inBasket}
   382→   382→      className={`mt-4 flex w-full items-center justify-center gap-2 border py-2.5 font-mono text-[10px] uppercase tracking-[0.18em] transition-all ${
   383→   383→        inBasket
   384→   384→          ? "border-[#00FF94] bg-[#00FF94]/10 text-[#00FF94]"
   385→   385→          : currencyMode === "INTL"
   386→   386→            ? "border-foreground/60 text-foreground hover:border-[#FF4500] hover:text-[#FF4500]"
   387→   387→            : popular
   388→   388→              ? "border-[#FF4500] bg-[#FF4500] text-black hover:opacity-80"
   389→   389→              : "border-foreground/60 text-foreground hover:border-[#FF4500] hover:text-[#FF4500]"
   390→   390→      }`}
   391→   391→    >
   392→   392→      {inBasket ? (
   393→   393→        <>
   394→   394→          <Check className="h-3 w-3" /> IN BASKET
   395→   395→        </>
   396→   396→      ) : (
   397→   397→        <>
   398→   398→          <Plus className="h-3 w-3" /> {buttonLabel}
