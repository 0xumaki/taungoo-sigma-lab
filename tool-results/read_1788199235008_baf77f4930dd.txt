    14→function useLiveTimestamp() {
    15→  const [ts, setTs] = React.useState("T+04:17:22");
    16→  React.useEffect(() => {
    17→    let sec = 4 * 3600 + 17 * 60 + 22;
    18→    const id = setInterval(() => {
    19→      sec += 1;
    20→      const h = String(Math.floor(sec / 3600)).padStart(2, "0");
    21→      const m = String(Math.floor((sec % 3600) / 60)).padStart(2, "0");
    22→      const s = String(sec % 60).padStart(2, "0");
    23→      setTs(`T+${h}:${m}:${s}`);
    24→    }, 1000);
    25→    return () => clearInterval(id);
    26→  }, []);
    27→  return ts;
    28→}

    46→function useCountUp(target: number, duration = 1500, delay = 0) {
    47→  const [val, setVal] = React.useState(0);
    48→  React.useEffect(() => {
    49→    let raf: number;
    50→    const start = performance.now() + delay;
    51→    const tick = (now: number) => {
    52→      if (now < start) { raf = requestAnimationFrame(tick); return; }
    53→      const t = Math.min(1, (now - start) / duration);
    54→      const eased = 1 - Math.pow(1 - t, 3);
    55→      setVal(target * eased);
    56→      if (t < 1) raf = requestAnimationFrame(tick);
    57→    };
    58→    raf = requestAnimationFrame(tick);
    59→    return () => cancelAnimationFrame(raf);
    60→  }, [target, duration, delay]);
    61→  return val;
    62→}

    64→/**
    65→ * useTextScramble — cyberpunk decoder effect.
    66→ * Returns a string that starts as random chars and resolves to the target.
    67→ */
    68→const SCRAMBLE_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZΣ0123456789";
    69→function randomScramble(len: number) {
    70→  let s = "";
    71→  for (let i = 0; i < len; i++) s += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
    72→  return s;
    73→}
    74→function useTextScramble(target: string, duration = 1200, delay = 0) {
    75→  // Initialize with the TARGET string (deterministic) so SSR output matches the
    76→  // first client render — the scramble then starts inside useEffect (client-only),
    77→  // which fires immediately after hydration. Fixes the hydration mismatch warning.
    78→  const [text, setText] = React.useState(() => target);
    79→  React.useEffect(() => {
    80→    let raf: number;
    81→    const start = performance.now() + delay;
    82→    const tick = (now: number) => {
    83→      if (now < start) {
    84→        // Update scramble rapidly while waiting for delay to pass
    85→        setText(randomScramble(target.length));
    86→        raf = requestAnimationFrame(tick);
    87→        return;
    88→      }
    89→      const t = Math.min(1, (now - start) / duration);
    90→      // Number of chars resolved = ceil(t * length)
    91→      const resolved = Math.ceil(t * target.length);
    92→      let result = "";
    93→      for (let i = 0; i < target.length; i++) {
    94→        if (i < resolved || target[i] === " ") result += target[i];
    95→        else result += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
    96→      }
    97→      setText(result);
    98→      if (t < 1) raf = requestAnimationFrame(tick);
    99→    };
   100→    raf = requestAnimationFrame(tick);
   101→    return () => cancelAnimationFrame(raf);
   102→  }, [target, duration, delay]);
   103→  return text;
   104→}

   118→/**
   119→ * useMouseParallax — returns spring-smoothed x/y motion values (in px) for a
   120→ * subtle parallax effect based on mouse position relative to viewport center.
   121→ * strength: how many px to move at max (e.g. 20 = ±20px)
   122→ */
   123→function useMouseParallax(strength = 20) {
   124→  const x = useMotionValue(0);
   125→  const y = useMotionValue(0);
   126→  const sx = useSpring(x, { stiffness: 80, damping: 20, restDelta: 0.1 });
   127→  const sy = useSpring(y, { stiffness: 80, damping: 20, restDelta: 0.1 });
   128→  React.useEffect(() => {
   129→    if (typeof window === "undefined") return;
   130→    const finePointer = window.matchMedia("(pointer: fine)").matches;
   131→    if (!finePointer) return;
   132→    // Respect prefers-reduced-motion: skip mouse parallax entirely.
   133→    // (Hero already disables scroll parallax via useReducedMotion in render.)
   134→    const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   135→    if (reducedMotion) return;
   136→    const onMove = (e: MouseEvent) => {
   137→      const cx = window.innerWidth / 2;
   138→      const cy = window.innerHeight / 2;
   139→      // Normalize to [-1, 1]
   140→      const nx = (e.clientX - cx) / cx;
   141→      const ny = (e.clientY - cy) / cy;
   142→      x.set(nx * strength);
   143→      y.set(ny * strength);
   144→    };
   145→    window.addEventListener("mousemove", onMove);
   146→    return () => window.removeEventListener("mousemove", onMove);
   147→  }, [strength, x, y]);
   148→  return { x: sx, y: sy };
   149→}

   151→/**
   152→ * useMagnetic — returns a callback ref that attaches a magnetic-cursor effect.
   153→ * Apply to a button/anchor for subtle cursor attraction (max `strength` px pull).
   154→ * Respects prefers-reduced-motion (skips entirely) + touch devices (pointer: coarse).
   155→ * Uses a callback ref pattern so it works for both always-mounted elements AND
   156→ * conditionally-rendered ones (e.g., a multi-step form's submit button that only
   157→ * appears on the last step). Exported so other beta components (Contact submit)
   158→ * can reuse the same px-based magnetic feel as the Hero CTA.
   159→ */
   160→export function useMagnetic<T extends HTMLElement = HTMLAnchorElement>(strength = 6) {
   161→  const [el, setEl] = React.useState<T | null>(null);
   162→  const ref = React.useCallback((node: T | null) => setEl(node), []);
   163→  React.useEffect(() => {
   164→    if (!el) return;
   165→    if (typeof window === "undefined") return;
   166→    const finePointer = window.matchMedia("(pointer: fine)").matches;
   167→    if (!finePointer) return;
   168→    const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   169→    if (reducedMotion) return;
   170→    // PERF (LOOP-1-LH): cache the button's bounding rect on mouseenter (when
   171→    // the cursor first enters the magnetic field), invalidate on scroll/resize.
   172→    // Previously, every window mousemove forced a getBoundingClientRect() layout
   173→    // reflow — and this is a WINDOW-level listener (fires for every mouse move
   174→    // anywhere on the page, not just over the button). That's the worst-case
   175→    // layout-thrash pattern: 60+ forced layouts/sec while the cursor is anywhere.
   176→    let r: DOMRect | null = null;
   177→    const refreshRect = () => { r = el.getBoundingClientRect(); };
   178→    const invalidateRect = () => { r = null; };
   179→    const onMove = (e: MouseEvent) => {
   180→      // Lazy refresh if cache was invalidated.
   181→      if (!r) refreshRect();
   182→      const rect = r as DOMRect;
   183→      const cx = rect.left + rect.width / 2;
   184→      const cy = rect.top + rect.height / 2;
   185→      const dx = (e.clientX - cx) / rect.width;
   186→      const dy = (e.clientY - cy) / rect.height;
   187→      // Only attract when cursor is within 1.5x button size
   188→      if (Math.abs(dx) < 1.5 && Math.abs(dy) < 1.5) {
   189→        el.style.transform = `translate(${dx * strength}px, ${dy * strength}px)`;
   190→      } else {
   191→        el.style.transform = "translate(0, 0)";
   192→      }
   193→    };
   194→    const onLeave = () => { el.style.transform = "translate(0, 0)"; };
   195→    el.addEventListener("mouseenter", refreshRect);
   196→    window.addEventListener("mousemove", onMove);
   197→    el.addEventListener("mouseleave", onLeave);
   198→    window.addEventListener("scroll", invalidateRect, { capture: true, passive: true });
   199→    window.addEventListener("resize", invalidateRect, { passive: true });
   200→    return () => {
   201→      el.removeEventListener("mouseenter", refreshRect);
   202→      window.removeEventListener("mousemove", onMove);
   203→      el.removeEventListener("mouseleave", onLeave);
   204→      window.removeEventListener("scroll", invalidateRect, { capture: true });
   205→      window.removeEventListener("resize", invalidateRect);
   206→      el.style.transform = "";
   207→    };
   208→  }, [el, strength]);
   209→  return ref;
   210→}

   212→/**
   213→ * useTypingText — types out text char-by-char, then blinks cursor.
   214→ * Loops after a pause.
   215→ */
   216→function useTypingText(text: string, speed = 50, startDelay = 0, pauseAfter = 2500) {
   217→  const [output, setOutput] = React.useState("");
   218→  const [done, setDone] = React.useState(false);
   219→  React.useEffect(() => {
   220→    let i = 0;
   221→    let timer: ReturnType<typeof setTimeout>;
   222→    let pauseTimer: ReturnType<typeof setTimeout>;
   223→    const startTimer = setTimeout(function type() {
   224→      if (i <= text.length) {
   225→        setOutput(text.slice(0, i));
   226→        setDone(i === text.length);
   227→        i++;
   228→        timer = setTimeout(type, speed);
   229→      } else {
   230→        // Pause then restart
   231→        pauseTimer = setTimeout(() => {
   232→          i = 0;
   233→          setOutput("");
   234→          setDone(false);
   235→          timer = setTimeout(type, speed);
   236→        }, pauseAfter);
   237→      }
   238→    }, startDelay);
   239→    return () => {
   240→      clearTimeout(startTimer);
   241→      clearTimeout(timer);
   242→      clearTimeout(pauseTimer);
   243→    };
   244→  }, [text, speed, startDelay, pauseAfter]);
   245→  return { output, done };
   246→}

   248→/**
   249→ * useLiveBars — returns an array of heights (0-100) that update on an interval.
   250→ * Used for the throughput monitor visualization.
   251→ *
   252→ * HYDRATION SAFETY: the initial state is deterministic (all bars at 50) so the
   253→ * SSR HTML matches the first client render. The randomization only starts inside
   254→ * useEffect (client-only) — same pattern as useTextScramble + useCountUp above.
   255→ * Without this guard, Math.random() in the useState initializer would produce
   256→ * different bar values on server vs client → hydration mismatch warning.
   257→ */
   258→function useLiveBars(count = 12, intervalMs = 1200) {
   259→  const [bars, setBars] = React.useState<number[]>(() =>
   260→    Array.from({ length: count }, () => 50)
   261→  );
   262→  React.useEffect(() => {
   263→    // First tick: seed real random values (client-only).
   264→    setBars(Array.from({ length: count }, () => 20 + Math.random() * 70));
   265→    const id = setInterval(() => {
   266→      // Skip oscilloscope churn when the tab is hidden — bars are decorative.
   267→      if (document.hidden) return;
   268→      setBars((prev) => {
   269→        // Shift left, push new value at end (oscilloscope-style)
   270→        const next = [...prev.slice(1), 20 + Math.random() * 75];
   271→        return next;
   272→      });
   273→    }, intervalMs);
   274→    return () => clearInterval(id);
   275→  }, [count, intervalMs]);
   276→  return bars;
   277→}
