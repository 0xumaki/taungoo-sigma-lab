    13→function useLiveTimestamp() {
    14→  const [ts, setTs] = React.useState("T+04:17:22");
    15→  React.useEffect(() => {
    16→    let sec = 4 * 3600 + 17 * 60 + 22;
    17→    const id = setInterval(() => {
    18→      sec += 1;
    19→      const h = String(Math.floor(sec / 3600)).padStart(2, "0");
    20→      const m = String(Math.floor((sec % 3600) / 60)).padStart(2, "0");
    21→      const s = String(sec % 60).padStart(2, "0");
    22→      setTs(`T+${h}:${m}:${s}`);
    23→    }, 1000);
    24→    return () => clearInterval(id);
    25→  }, []);
    26→  return ts;
    27→}

    42→function useCountUp(target: number, duration = 1500, delay = 0) {
    43→  const [val, setVal] = React.useState(0);
    44→  React.useEffect(() => {
    45→    let raf: number;
    46→    const start = performance.now() + delay;
    47→    const tick = (now: number) => {
    48→      if (now < start) { raf = requestAnimationFrame(tick); return; }
    49→      const t = Math.min(1, (now - start) / duration);
    50→      const eased = 1 - Math.pow(1 - t, 3);
    51→      setVal(target * eased);
    52→      if (t < 1) raf = requestAnimationFrame(tick);
    53→    };
    54→    raf = requestAnimationFrame(tick);
    55→    return () => cancelAnimationFrame(raf);
    56→  }, [target, duration, delay]);
    57→  return val;
    58→}

    60→/**
    61→ * useTextScramble — cyberpunk decoder effect.
    62→ * Returns a string that starts as random chars and resolves to the target.
    63→ */
    64→const SCRAMBLE_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZΣ0123456789";
    65→function randomScramble(len: number) {
    66→  let s = "";
    67→  for (let i = 0; i < len; i++) s += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
    68→  return s;
    69→}
    70→function useTextScramble(target: string, duration = 1200, delay = 0) {
    71→  // Initialize with the TARGET string (deterministic) so SSR output matches the
    72→  // first client render — the scramble then starts inside useEffect (client-only),
    73→  // which fires immediately after hydration. Fixes the hydration mismatch warning.
    74→  const [text, setText] = React.useState(() => target);
    75→  React.useEffect(() => {
    76→    let raf: number;
    77→    const start = performance.now() + delay;
    78→    const tick = (now: number) => {
    79→      if (now < start) {
    80→        // Update scramble rapidly while waiting for delay to pass
    81→        setText(randomScramble(target.length));
    82→        raf = requestAnimationFrame(tick);
    83→        return;
    84→      }
    85→      const t = Math.min(1, (now - start) / duration);
    86→      // Number of chars resolved = ceil(t * length)
    87→      const resolved = Math.ceil(t * target.length);
    88→      let result = "";
    89→      for (let i = 0; i < target.length; i++) {
    90→        if (i < resolved || target[i] === " ") result += target[i];
    91→        else result += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
    92→      }
    93→      setText(result);
    94→      if (t < 1) raf = requestAnimationFrame(tick);
    95→    };
    96→    raf = requestAnimationFrame(tick);
    97→    return () => cancelAnimationFrame(raf);
    98→  }, [target, duration, delay]);
    99→  return text;
   100→}

   114→/**
   115→ * useMouseParallax — returns spring-smoothed x/y motion values (in px) for a
   116→ * subtle parallax effect based on mouse position relative to viewport center.
   117→ * strength: how many px to move at max (e.g. 20 = ±20px)
   118→ */
   119→function useMouseParallax(strength = 20) {
   120→  const x = useMotionValue(0);
   121→  const y = useMotionValue(0);
   122→  const sx = useSpring(x, { stiffness: 80, damping: 20, restDelta: 0.1 });
   123→  const sy = useSpring(y, { stiffness: 80, damping: 20, restDelta: 0.1 });
   124→  React.useEffect(() => {
   125→    if (typeof window === "undefined") return;
   126→    const finePointer = window.matchMedia("(pointer: fine)").matches;
   127→    if (!finePointer) return;
   128→    // Respect prefers-reduced-motion: skip mouse parallax entirely.
   129→    // (Hero already disables scroll parallax via useReducedMotion in render.)
   130→    const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   131→    if (reducedMotion) return;
   132→    const onMove = (e: MouseEvent) => {
   133→      const cx = window.innerWidth / 2;
   134→      const cy = window.innerHeight / 2;
   135→      // Normalize to [-1, 1]
   136→      const nx = (e.clientX - cx) / cx;
   137→      const ny = (e.clientY - cy) / cy;
   138→      x.set(nx * strength);
   139→      y.set(ny * strength);
   140→    };
   141→    window.addEventListener("mousemove", onMove);
   142→    return () => window.removeEventListener("mousemove", onMove);
   143→  }, [strength, x, y]);
   144→  return { x: sx, y: sy };
   145→}

   147→/**
   148→ * useMagnetic — returns a callback ref that attaches a magnetic-cursor effect.
   149→ * Apply to a button/anchor for subtle cursor attraction (max `strength` px pull).
   150→ * Respects prefers-reduced-motion (skips entirely) + touch devices (pointer: coarse).
   151→ * Uses a callback ref pattern so it works for both always-mounted elements AND
   152→ * conditionally-rendered ones (e.g., a multi-step form's submit button that only
   153→ * appears on the last step). Exported so other beta components (Contact submit)
   154→ * can reuse the same px-based magnetic feel as the Hero CTA.
   155→ */
   156→export function useMagnetic<T extends HTMLElement = HTMLAnchorElement>(strength = 6) {
   157→  const [el, setEl] = React.useState<T | null>(null);
   158→  const ref = React.useCallback((node: T | null) => setEl(node), []);
   159→  React.useEffect(() => {
   160→    if (!el) return;
   161→    if (typeof window === "undefined") return;
   162→    const finePointer = window.matchMedia("(pointer: fine)").matches;
   163→    if (!finePointer) return;
   164→    const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   165→    if (reducedMotion) return;
   166→    const onMove = (e: MouseEvent) => {
   167→      const r = el.getBoundingClientRect();
   168→      const cx = r.left + r.width / 2;
   169→      const cy = r.top + r.height / 2;
   170→      const dx = (e.clientX - cx) / r.width;
   171→      const dy = (e.clientY - cy) / r.height;
   172→      // Only attract when cursor is within 1.5x button size
   173→      if (Math.abs(dx) < 1.5 && Math.abs(dy) < 1.5) {
   174→        el.style.transform = `translate(${dx * strength}px, ${dy * strength}px)`;
   175→      } else {
   176→        el.style.transform = "translate(0, 0)";
   177→      }
   178→    };
   179→    const onLeave = () => { el.style.transform = "translate(0, 0)"; };
   180→    window.addEventListener("mousemove", onMove);
   181→    el.addEventListener("mouseleave", onLeave);
   182→    return () => {
   183→      window.removeEventListener("mousemove", onMove);
   184→      el.removeEventListener("mouseleave", onLeave);
   185→      el.style.transform = "";
   186→    };
   187→  }, [el, strength]);
   188→  return ref;
   189→}

   191→/**
   192→ * useTypingText — types out text char-by-char, then blinks cursor.
   193→ * Loops after a pause.
   194→ */
   195→function useTypingText(text: string, speed = 50, startDelay = 0, pauseAfter = 2500) {
   196→  const [output, setOutput] = React.useState("");
   197→  const [done, setDone] = React.useState(false);
   198→  React.useEffect(() => {
   199→    let i = 0;
   200→    let timer: ReturnType<typeof setTimeout>;
   201→    let pauseTimer: ReturnType<typeof setTimeout>;
   202→    const startTimer = setTimeout(function type() {
   203→      if (i <= text.length) {
   204→        setOutput(text.slice(0, i));
   205→        setDone(i === text.length);
   206→        i++;
   207→        timer = setTimeout(type, speed);
   208→      } else {
   209→        // Pause then restart
   210→        pauseTimer = setTimeout(() => {
   211→          i = 0;
   212→          setOutput("");
   213→          setDone(false);
   214→          timer = setTimeout(type, speed);
   215→        }, pauseAfter);
   216→      }
   217→    }, startDelay);
   218→    return () => {
   219→      clearTimeout(startTimer);
   220→      clearTimeout(timer);
   221→      clearTimeout(pauseTimer);
   222→    };
   223→  }, [text, speed, startDelay, pauseAfter]);
   224→  return { output, done };
   225→}

   227→/**
   228→ * useLiveBars — returns an array of heights (0-100) that update on an interval.
   229→ * Used for the throughput monitor visualization.
   230→ *
   231→ * HYDRATION SAFETY: the initial state is deterministic (all bars at 50) so the
   232→ * SSR HTML matches the first client render. The randomization only starts inside
   233→ * useEffect (client-only) — same pattern as useTextScramble + useCountUp above.
   234→ * Without this guard, Math.random() in the useState initializer would produce
   235→ * different bar values on server vs client → hydration mismatch warning.
   236→ */
   237→function useLiveBars(count = 12, intervalMs = 1200) {
   238→  const [bars, setBars] = React.useState<number[]>(() =>
   239→    Array.from({ length: count }, () => 50)
   240→  );
   241→  React.useEffect(() => {
   242→    // First tick: seed real random values (client-only).
   243→    setBars(Array.from({ length: count }, () => 20 + Math.random() * 70));
   244→    const id = setInterval(() => {
   245→      setBars((prev) => {
   246→        // Shift left, push new value at end (oscilloscope-style)
   247→        const next = [...prev.slice(1), 20 + Math.random() * 75];
   248→        return next;
   249→      });
   250→    }, intervalMs);
   251→    return () => clearInterval(id);
   252→  }, [count, intervalMs]);
   253→  return bars;
   254→}
