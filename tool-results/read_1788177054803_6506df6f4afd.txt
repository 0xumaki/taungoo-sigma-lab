    13→function useLiveTimestamp() {
    14→  const [ts, setTs] = React.useState("T+04:17:22");
    15→  React.useEffect(() => {
    16→    let sec = 4 * 3600 + 17 * 60 + 22;
    17→    const id = setInterval(() => {
    18→      sec += 1;
    19→      const h = String(Math.floor(sec / 3600)).padStart(2, "0");
    20→      const m = String(Math.floor((sec % 3600) / 60)).padStart(2, "0");
    21→      const s = String(sec % 60).padStart(2, "0");
    22→      setTs(`T+${h}:${m}:${s}`);
    23→    }, 1000);
    24→    return () => clearInterval(id);
    25→  }, []);
    26→  return ts;
    27→}

    42→function useCountUp(target: number, duration = 1500, delay = 0) {
    43→  const [val, setVal] = React.useState(0);
    44→  React.useEffect(() => {
    45→    let raf: number;
    46→    const start = performance.now() + delay;
    47→    const tick = (now: number) => {
    48→      if (now < start) { raf = requestAnimationFrame(tick); return; }
    49→      const t = Math.min(1, (now - start) / duration);
    50→      const eased = 1 - Math.pow(1 - t, 3);
    51→      setVal(target * eased);
    52→      if (t < 1) raf = requestAnimationFrame(tick);
    53→    };
    54→    raf = requestAnimationFrame(tick);
    55→    return () => cancelAnimationFrame(raf);
    56→  }, [target, duration, delay]);
    57→  return val;
    58→}

    60→/**
    61→ * useTextScramble — cyberpunk decoder effect.
    62→ * Returns a string that starts as random chars and resolves to the target.
    63→ */
    64→const SCRAMBLE_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZΣ0123456789";
    65→function randomScramble(len: number) {
    66→  let s = "";
    67→  for (let i = 0; i < len; i++) s += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
    68→  return s;
    69→}
    70→function useTextScramble(target: string, duration = 1200, delay = 0) {
    71→  // Start with a scrambled string so user sees decode effect from t=0
    72→  const [text, setText] = React.useState(() => randomScramble(target.length));
    73→  React.useEffect(() => {
    74→    let raf: number;
    75→    const start = performance.now() + delay;
    76→    const tick = (now: number) => {
    77→      if (now < start) {
    78→        // Update scramble rapidly while waiting for delay to pass
    79→        setText(randomScramble(target.length));
    80→        raf = requestAnimationFrame(tick);
    81→        return;
    82→      }
    83→      const t = Math.min(1, (now - start) / duration);
    84→      // Number of chars resolved = ceil(t * length)
    85→      const resolved = Math.ceil(t * target.length);
    86→      let result = "";
    87→      for (let i = 0; i < target.length; i++) {
    88→        if (i < resolved || target[i] === " ") result += target[i];
    89→        else result += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
    90→      }
    91→      setText(result);
    92→      if (t < 1) raf = requestAnimationFrame(tick);
    93→    };
    94→    raf = requestAnimationFrame(tick);
    95→    return () => cancelAnimationFrame(raf);
    96→  }, [target, duration, delay]);
    97→  return text;
    98→}

   112→/**
   113→ * useMouseParallax — returns spring-smoothed x/y motion values (in px) for a
   114→ * subtle parallax effect based on mouse position relative to viewport center.
   115→ * strength: how many px to move at max (e.g. 20 = ±20px)
   116→ */
   117→function useMouseParallax(strength = 20) {
   118→  const x = useMotionValue(0);
   119→  const y = useMotionValue(0);
   120→  const sx = useSpring(x, { stiffness: 80, damping: 20, restDelta: 0.1 });
   121→  const sy = useSpring(y, { stiffness: 80, damping: 20, restDelta: 0.1 });
   122→  React.useEffect(() => {
   123→    if (typeof window === "undefined") return;
   124→    const finePointer = window.matchMedia("(pointer: fine)").matches;
   125→    if (!finePointer) return;
   126→    const onMove = (e: MouseEvent) => {
   127→      const cx = window.innerWidth / 2;
   128→      const cy = window.innerHeight / 2;
   129→      // Normalize to [-1, 1]
   130→      const nx = (e.clientX - cx) / cx;
   131→      const ny = (e.clientY - cy) / cy;
   132→      x.set(nx * strength);
   133→      y.set(ny * strength);
   134→    };
   135→    window.addEventListener("mousemove", onMove);
   136→    return () => window.removeEventListener("mousemove", onMove);
   137→  }, [strength, x, y]);
   138→  return { x: sx, y: sy };
   139→}

   141→/**
   142→ * useMagnetic — returns a callback ref that attaches a magnetic-cursor effect.
   143→ * Apply to a button/anchor for subtle cursor attraction (max `strength` px pull).
   144→ * Respects prefers-reduced-motion (skips entirely) + touch devices (pointer: coarse).
   145→ * Uses a callback ref pattern so it works for both always-mounted elements AND
   146→ * conditionally-rendered ones (e.g., a multi-step form's submit button that only
   147→ * appears on the last step). Exported so other beta components (Contact submit)
   148→ * can reuse the same px-based magnetic feel as the Hero CTA.
   149→ */
   150→export function useMagnetic<T extends HTMLElement = HTMLAnchorElement>(strength = 6) {
   151→  const [el, setEl] = React.useState<T | null>(null);
   152→  const ref = React.useCallback((node: T | null) => setEl(node), []);
   153→  React.useEffect(() => {
   154→    if (!el) return;
   155→    if (typeof window === "undefined") return;
   156→    const finePointer = window.matchMedia("(pointer: fine)").matches;
   157→    if (!finePointer) return;
   158→    const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   159→    if (reducedMotion) return;
   160→    const onMove = (e: MouseEvent) => {
   161→      const r = el.getBoundingClientRect();
   162→      const cx = r.left + r.width / 2;
   163→      const cy = r.top + r.height / 2;
   164→      const dx = (e.clientX - cx) / r.width;
   165→      const dy = (e.clientY - cy) / r.height;
   166→      // Only attract when cursor is within 1.5x button size
   167→      if (Math.abs(dx) < 1.5 && Math.abs(dy) < 1.5) {
   168→        el.style.transform = `translate(${dx * strength}px, ${dy * strength}px)`;
   169→      } else {
   170→        el.style.transform = "translate(0, 0)";
   171→      }
   172→    };
   173→    const onLeave = () => { el.style.transform = "translate(0, 0)"; };
   174→    window.addEventListener("mousemove", onMove);
   175→    el.addEventListener("mouseleave", onLeave);
   176→    return () => {
   177→      window.removeEventListener("mousemove", onMove);
   178→      el.removeEventListener("mouseleave", onLeave);
   179→      el.style.transform = "";
   180→    };
   181→  }, [el, strength]);
   182→  return ref;
   183→}

   185→/**
   186→ * useTypingText — types out text char-by-char, then blinks cursor.
   187→ * Loops after a pause.
   188→ */
   189→function useTypingText(text: string, speed = 50, startDelay = 0, pauseAfter = 2500) {
   190→  const [output, setOutput] = React.useState("");
   191→  const [done, setDone] = React.useState(false);
   192→  React.useEffect(() => {
   193→    let i = 0;
   194→    let timer: ReturnType<typeof setTimeout>;
   195→    let pauseTimer: ReturnType<typeof setTimeout>;
   196→    const startTimer = setTimeout(function type() {
   197→      if (i <= text.length) {
   198→        setOutput(text.slice(0, i));
   199→        setDone(i === text.length);
   200→        i++;
   201→        timer = setTimeout(type, speed);
   202→      } else {
   203→        // Pause then restart
   204→        pauseTimer = setTimeout(() => {
   205→          i = 0;
   206→          setOutput("");
   207→          setDone(false);
   208→          timer = setTimeout(type, speed);
   209→        }, pauseAfter);
   210→      }
   211→    }, startDelay);
   212→    return () => {
   213→      clearTimeout(startTimer);
   214→      clearTimeout(timer);
   215→      clearTimeout(pauseTimer);
   216→    };
   217→  }, [text, speed, startDelay, pauseAfter]);
   218→  return { output, done };
   219→}

   222→ * useLiveBars — returns an array of heights (0-100) that update on an interval.
   223→ * Used for the throughput monitor visualization.

   225→function useLiveBars(count = 12, intervalMs = 1200) {
   226→  const [bars, setBars] = React.useState<number[]>(() =>
   227→    Array.from({ length: count }, () => 20 + Math.random() * 70)
   228→  );
   229→  React.useEffect(() => {
   230→    const id = setInterval(() => {
   231→      setBars((prev) => {
   232→        // Shift left, push new value at end (oscilloscope-style)
   233→        const next = [...prev.slice(1), 20 + Math.random() * 75];
   234→        return next;
   235→      });
   236→    }, intervalMs);
   237→    return () => clearInterval(id);
   238→  }, [count, intervalMs]);
   239→  return bars;
   240→}
