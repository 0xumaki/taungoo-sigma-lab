    91→export function ExperienceShell() {
    92→  const { view, phase, navigate, setPhase, visitToken } = useSigmaStore();
    93→  // Auto-enable SFX on first user interaction (both Sigma and Alpha modes)
    94→  useSigmaSound();
    95→  const [renderedView, setRenderedView] = React.useState<SectionId>(view);
    96→  const [flashAccent, setFlashAccent] = React.useState<string>("#FFFFFF");
    97→  const [booting, setBooting] = React.useState(false);
    98→  const [onboarding, setOnboarding] = React.useState(false);
    99→  const [cmdOpen, setCmdOpen] = React.useState(false);
   100→  const [mcActive, setMcActive] = React.useState(false);
   101→  const [tourActive, setTourActive] = React.useState(false);
   102→  const [tourPaused, setTourPaused] = React.useState(false);
   103→  const [tourIndex, setTourIndex] = React.useState(0);
   104→  const [mode, setMode] = React.useState<"sigma" | "alpha" | "beta">("beta");
   105→
   106→  // Deep-link + boot screen + onboarding logic — single effect, runs once after mount
   107→  React.useEffect(() => {
   108→    if (typeof window === "undefined") return;
   109→    const params = new URLSearchParams(window.location.search);
   110→    const s = params.get("s");
   111→    if (s) {
   112→      const id = `s${s.padStart(2, "0")}` as SectionId;
   113→      // Defer to next tick to avoid hydration race condition
   114→      const t = setTimeout(() => {
   115→        useSigmaStore.getState().boot(id);
   116→        setRenderedView(id);
   117→      }, 0);
   118→      return () => clearTimeout(t); // deep-links skip the boot screen + onboarding
   119→    }
   120→    // Only show boot screen on first load or full page reload — NOT when
   121→    // returning from detail pages. We detect this by checking if the page
   122→    // was loaded via browser navigation (back/forward) vs a full reload.
   123→    // performance.navigation.type === 0 = normal navigation (could be back from detail)
   124→    // performance.navigation.type === 1 = reload (show boot)
   125→    // If there's a saved scroll position, it means we're returning from a detail page
   126→    // CRITICAL: Check BOTH alpha_scroll_position AND beta_scroll_position — Beta mode
   127→    // saves to beta_scroll_position, so without this check, Beta users see the boot
   128→    // screen when returning from detail pages (bug reported in Stage 70).
   129→    const isReturning =
   130→      sessionStorage.getItem("alpha_scroll_position") !== null ||
   131→      sessionStorage.getItem("beta_scroll_position") !== null;
   132→    const navEntry = performance.getEntriesByType("navigation")[0] as PerformanceNavigationTiming | undefined;
   133→    const isReload = navEntry?.type === "reload";
   134→
   135→    // Determine the mode early — Beta mode is a clean enterprise mode that should
   136→    // NEVER show the brutalist SigmaBoot screen. Only Sigma mode gets the boot screen.
   137→    // We read the saved mode from localStorage (same logic as SigmaModeSwitcher).
   138→    let savedMode: "sigma" | "alpha" | "beta" | null = null;
   139→    try {
   140→      const sm = localStorage.getItem("sigma-mode");
   141→      if (sm === "sigma" || sm === "alpha" || sm === "beta") savedMode = sm;
   142→    } catch { /* ignore */ }
   143→    // If no saved mode, default is "beta" (per Stage 68 user spec)
   144→    const effectiveMode = savedMode || "beta";
   145→
   146→    if (effectiveMode === "beta") {
   147→      // Beta mode NEVER shows the boot screen — it's a clean enterprise mode
   148→      setBooting(false);
   149→    } else if (!isReturning && !isReload) {
   150→      // First visit to the site (not returning from detail, not a reload)
   151→      // → show boot screen
   152→      setBooting(true);
   153→    } else if (isReload) {
   154→      // Full page reload → show boot screen
   155→      setBooting(true);
   156→    }
   157→    // If isReturning → skip boot screen (coming back from detail page)
   158→
   159→    try {
   160→      // show onboarding only once per browser (localStorage)
   161→      // Beta mode skips onboarding entirely (clean enterprise mode — no Sigma/Alpha tutorials)
   162→      const onboarded = localStorage.getItem("sigma_onboarded");
   163→      if (!onboarded && !isReturning && effectiveMode !== "beta") {
   164→        // delay onboarding until after boot screen
   165→        const ob = setTimeout(() => setOnboarding(true), 4000);
   166→        return () => clearTimeout(ob);
   167→      }
   168→    } catch {
   169→      // sessionStorage/localStorage may be unavailable (private mode) — skip
   170→    }
   171→  }, []);
   172→
   173→  const handleBootDone = React.useCallback(() => {
   174→    try {
   175→      sessionStorage.setItem("sigma_booted", "1");
   176→    } catch {
   177→      // ignore
   178→    }
   179→    setBooting(false);
   180→  }, []);
   181→
   182→  // Safety: if boot screen stays for >6s, force-clear it (prevents stuck blank page)
   183→  React.useEffect(() => {
   184→    if (!booting) return;
   185→    const t = setTimeout(() => {
   186→      setBooting(false);
   187→      try {
   188→        sessionStorage.setItem("sigma_booted", "1");
   189→      } catch {
   190→        // ignore
   191→      }
   192→    }, 6000);
   193→    return () => clearTimeout(t);
   194→  }, [booting]);
   195→
   196→  // Command palette: Cmd/Ctrl+K to open, also '/' as a secondary trigger
   197→  React.useEffect(() => {
   198→    const onKey = (e: KeyboardEvent) => {
   199→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
   200→        e.preventDefault();
   201→        setCmdOpen((o) => !o);
   202→      } else if (e.key === "/" && !cmdOpen) {
   203→        // only trigger '/' when not already typing in an input
   204→        const t = e.target as HTMLElement;
   205→        if (t && (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.isContentEditable)) return;
   206→        e.preventDefault();
   207→        setCmdOpen(true);
   208→      }
   209→    };
   210→    window.addEventListener("keydown", onKey);
   211→    return () => window.removeEventListener("keydown", onKey);
   212→  }, [cmdOpen]);
   213→
   214→  const panelsRef = React.useRef<(HTMLDivElement | null)[]>([]);
   215→  const overlayRef = React.useRef<HTMLDivElement>(null);
   216→  const flashRef = React.useRef<HTMLDivElement>(null);
   217→  const labelRef = React.useRef<HTMLDivElement>(null);
   218→  const rootRef = React.useRef<HTMLDivElement>(null);
   219→
   220→  // Keyboard navigation — DISABLED when typing in form inputs (S10 Access, S08 search, etc.)
   221→  // ALSO DISABLED in Beta mode (user request: Alpha and Sigma key bindings should not work in Beta)
   222→  React.useEffect(() => {
   223→    const onKey = (e: KeyboardEvent) => {
   224→      // Skip ALL key bindings when in Beta mode
   225→      if (mode === "beta") return;
   226→      // Skip ALL key bindings when user is typing in an input/textarea/contenteditable
   227→      const target = e.target as HTMLElement;
   228→      if (target && (target.tagName === "INPUT" || target.tagName === "TEXTAREA" || target.isContentEditable)) {
   229→        return; // Let the user type without triggering navigation
   230→      }
   231→      if (useSigmaStore.getState().phase !== "idle") return;
   232→      const cur = useSigmaStore.getState().view;
   233→      if (e.key === "Escape") {
   234→        if (cur !== "map") { navigate("map"); sigmaSound.play("close"); }
   235→      } else if (e.key === "m" || e.key === "M") {
   236→        if (cur !== "map") { navigate("map"); sigmaSound.play("close"); }
   237→      } else if (e.key === "ArrowRight") {
   238→        navigate(nextSection(cur)); sigmaSound.play("click");
   239→      } else if (e.key === "ArrowLeft") {
   240→        navigate(prevSection(cur)); sigmaSound.play("click");
   241→      } else if (e.key >= "0" && e.key <= "9") {
   242→        const n = parseInt(e.key, 10);
   243→        const target =
   244→          n === 0 ? "map" : (`s${String(n).padStart(2, "0")}` as SectionId);
   245→        if (target !== cur) { navigate(target); sigmaSound.play("click"); }
   246→      } else if (e.key === "-" || e.key === "_") {
   247→        // s10 (the "-" key is next to 0 on the keyboard)
   248→        if (cur !== "s10") { navigate("s10"); sigmaSound.play("click"); }
   249→      } else if (e.key === "=" || e.key === "+") {
   250→        // s11 (the "=" key is next to "-" on the keyboard)
   251→        if (cur !== "s11") { navigate("s11"); sigmaSound.play("click"); }
   252→      } else if (e.key === "c" || e.key === "C") {
   253→        // MC mode toggle
   254→        const t = e.target as HTMLElement;
   255→        if (t && (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.isContentEditable)) return;
   256→        e.preventDefault();
   257→        setMcActive((a) => !a);
   258→      } else if (e.key === "t" || e.key === "T") {
   259→        // Tour toggle
   260→        const t = e.target as HTMLElement;
   261→        if (t && (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.isContentEditable)) return;
   262→        e.preventDefault();
   263→        if (tourActive) {
   264→          setTourPaused((p) => !p);
   265→        } else {
   266→          // start tour from current sector
   267→          const curIdx = SECTIONS.findIndex((s) => s.id === cur);
   268→          setTourIndex(curIdx >= 0 ? curIdx : 0);
   269→          setTourActive(true);
   270→          setTourPaused(false);
   271→        }
   272→      }
   273→    };
   274→    window.addEventListener("keydown", onKey);
   275→    return () => window.removeEventListener("keydown", onKey);
   276→  }, [navigate, tourActive, mode]);
   277→
   278→  // Safety: if phase stays "covering" for >4s (e.g. GSAP interrupted), force-reset to idle
   279→  React.useEffect(() => {
   280→    if (phase !== "covering") return;
   281→    const t = setTimeout(() => {
   282→      if (useSigmaStore.getState().phase === "covering") {
   283→        setPhase("idle");
   284→      }
   285→    }, 4000);
   286→    return () => clearTimeout(t);
   287→  }, [phase, setPhase]);
   288→
   289→  // Tour mode: auto-advance through sectors every 6s
   290→  React.useEffect(() => {
   291→    if (!tourActive || tourPaused) return;
   292→    if (phase !== "idle") return;
   293→    const SECTOR_DURATION = 6000;
   294→    const startTs = Date.now();
   295→    const interval = setInterval(() => {
   296→      if (Date.now() - startTs >= SECTOR_DURATION) {
   297→        setTourIndex((idx) => {
   298→          const next = (idx + 1) % SECTIONS.length;
   299→          navigate(SECTIONS[next].id);
   300→          sigmaSound.play("transition");
   301→          return next;
   302→        });
   303→      }
   304→    }, 1000);
   305→    return () => clearInterval(interval);
   306→  }, [tourActive, tourPaused, phase, navigate]);
   307→
   308→  // Run transition timeline on phase change — uses useEffect for reliability
   309→  React.useEffect(() => {
   310→    if (phase !== "covering") return;
   311→    const dest = useSigmaStore.getState().view;
   312→    const destMeta = getSection(dest);
   313→    setFlashAccent(destMeta.accent);
   314→    sigmaSound.play("transition");
   315→    // play ambient tone tuned to the destination sector's accent color
   316→    if (dest !== "map") {
   317→      sigmaSound.playAmbient(destMeta.accent);
   318→    }
   319→
   320→    // Check for reduced motion preference — if active, skip the animation and just swap
   321→    const prefersReduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   322→    if (prefersReduced) {
   323→      setRenderedView(dest);
   324→      setPhase("idle");
   325→      return;
   326→    }
   327→
   328→    const panels = panelsRef.current.filter(Boolean) as HTMLDivElement[];
   329→    const overlay = overlayRef.current;
   330→    const flash = flashRef.current;
   331→    const label = labelRef.current;
   332→
   333→    if (panels.length === 0 || !overlay) {
   334→      // no panels yet — just swap immediately
   335→      setRenderedView(dest);
   336→      setPhase("idle");
   337→      return;
   338→    }
   339→
   340→    let done = false;
   341→    const finish = () => {
   342→      if (done) return;
   343→      done = true;
   344→      setPhase("idle");
   345→    };
   346→
   347→    const tl = gsap.timeline({
   348→      onComplete: finish,
   349→      onReverseComplete: finish,
   350→    });
   351→
   352→    // 1. COVER — panels slam down, staggered from top
   353→    tl.set(panels, {
   354→      scaleY: 0,
   355→      transformOrigin: "top",
   356→      backgroundColor: destMeta.accent,
   357→    });
   358→    tl.set(overlay, { pointerEvents: "auto", zIndex: 90 });
   359→    tl.to(panels, {
   360→      scaleY: 1,
   361→      duration: 0.42,
   362→      ease: "power3.in",
   363→      stagger: { each: 0.04, from: "start" },
   364→    });
   365→
   366→    // 2. MIDPOINT — swap the rendered view + flash + label fly-through
   367→    tl.call(
   368→      () => {
   369→        setRenderedView(dest);
   370→        if (flash) {
   371→          gsap.fromTo(
   372→            flash,
   373→            { opacity: 1 },
   374→            { opacity: 0, duration: 0.5, ease: "power2.out" }
   375→          );
   376→        }
   377→        // Glitch flicker — fire a 180ms clip-path slice glitch on the overlay
   378→        // (sigma mode signature transition polish, reduced-motion safe via CSS)
   379→        const glitchEl = overlay?.querySelector("[data-glitch-layer]");
   380→        if (glitchEl) {
   381→          glitchEl.classList.remove("sigma-transition-glitch");
   382→          // Force reflow so the animation re-triggers on every transition
   383→          void (glitchEl as HTMLElement).offsetWidth;
   384→          glitchEl.classList.add("sigma-transition-glitch");
   385→        }
   386→      },
   387→      [],
   388→      ">-0.05"
   389→    );
   390→
   391→    // sector label flies through during slam cover transition.
   392→    // Set the label TEXT dynamically so it shows the DESTINATION sector code
   393→    // (including "00" for map — which wasn't showing before because the label
   394→    // was bound to the rendered view's meta, not the destination's).
   395→    if (label) {
   396→      // Step 1: Set the text content to the destination sector code
   397→      tl.call(() => {
   398→        label.textContent = destMeta.shortCode;
   399→      }, [], "<");
   400→      // Step 2: Set visibility:visible + initial position (BEFORE animating opacity,
   401→      // so the element is visible when opacity starts increasing)
   402→      tl.set(label, { visibility: "visible", opacity: 0, x: -120 }, "<");
   403→      // Step 3: Animate opacity 0→1 + x: -120→0 (fly in from left)
   404→      tl.to(
   405→        label,
   406→        { opacity: 1, x: 0, duration: 0.28, ease: "power3.out" },
   407→        "<"
   408→      );
   409→      // Step 4: Animate opacity 1→0 + x: 0→120 (fly out to right)
   410→      tl.to(
   411→        label,
   412→        { opacity: 0, x: 120, duration: 0.28, ease: "power3.in" },
   413→        ">+0.04"
   414→      );
   415→    }
   416→
   417→    // 3. REVEAL — panels retract from bottom, staggered
   418→    tl.set(panels, { transformOrigin: "bottom" });
   419→    tl.to(panels, {
   420→      scaleY: 0,
   421→      duration: 0.46,
   422→      ease: "power3.out",
   423→      stagger: { each: 0.035, from: "start" },
   424→      clearProps: "transform,backgroundColor",
   425→    });
   426→
   427→    tl.set(overlay, { pointerEvents: "none" });
   428→    // CRITICAL: explicitly set opacity to 0 AND visibility to hidden (do NOT use clearProps — clearProps
   429→    // reverts to CSS default opacity:1, which makes the number persist on screen).
   430→    // visibility:hidden is the FINAL safety net — even if opacity fails for any reason,
   431→    // the element will still be completely invisible and non-interactive.
   432→    // This was the root cause of the "center number keeps displaying" bug.
   433→    if (label) {
   434→      tl.set(label, { opacity: 0, x: 0, visibility: "hidden" });
   435→    }
   436→
   437→    // Safety: if timeline doesn't complete in 3s, force-finish
   438→    const safety = setTimeout(() => {
   439→      if (!done) {
   440→        tl.kill();
   441→        setRenderedView(dest);
   442→        finish();
   443→      }
   444→    }, 3000);
   445→
   446→    return () => {
   447→      clearTimeout(safety);
   448→    };
   449→  }, [phase, setPhase]);
   450→
   451→  const meta = getSection(renderedView);
   452→
   453→  return (
   454→    <div
   455→      ref={rootRef}
   456→      suppressHydrationWarning
   457→      className={`relative fixed inset-0 overflow-hidden bg-background ${
   458→        mode === "beta" ? "" : mode === "sigma" ? "sigma-mode sigma-noise sigma-vignette" : "sigma-noise sigma-vignette"
   459→      }`}
   460→    >
   461→      {/* Skip-to-content link — RENDERED FIRST (DOM order) so it's the first
   462→          focusable element on the page. Visually hidden until focused
   463→          (keyboard-only). Targets #top in the Hero. Beta-only. The
   464→          visually-identical duplicate inside BetaInterface was removed to
   465→          avoid double-rendering + double-announce. WCAG 2.1 SC 2.4.1. */}
   466→      {mode === "beta" && (
   467→        <a href="#top" className="bs-skip-link">
   468→          Skip to content
   469→        </a>
   470→      )}
   471→
   472→      {/* Persistent layered background — HIDDEN in Beta mode (clean enterprise) */}
   473→      {mode !== "beta" && <div className="sigma-grid pointer-events-none absolute inset-0 opacity-60" />}
   474→
   475→      {/* Cursor spotlight (desktop only) — HIDDEN in Beta mode */}
   476→      {mode !== "beta" && <SigmaSpotlight />}
   477→
   478→      {/* Mode switcher (Σ / α / β) — floating, repositioned to top-right to avoid nav collision */}
   479→      {mode === "beta" && (
   480→        <div className="fixed right-4 top-20 z-[95]">
   481→          <SigmaModeSwitcher mode={mode} onModeChange={setMode} floating />
   482→        </div>
   483→      )}
   484→      {mode !== "beta" && <SigmaModeSwitcher mode={mode} onModeChange={setMode} />}
   485→
   486→      {/* Rendered view — Sigma mode (map-based), Alpha mode (brutalist scroll), or Beta mode (enterprise) */}
   487→      {mode === "sigma" ? (
   488→        <div
   489→          key={`${renderedView}-${visitToken}`}
   490→          className="absolute inset-0"
   491→          style={{ paddingTop: 32, paddingLeft: 32, paddingRight: 32, paddingBottom: 28 }}
   492→        >
   493→          <div className="relative h-full w-full">{renderView(renderedView)}</div>
   494→        </div>
   495→      ) : mode === "alpha" ? (
   496→        <AlphaInterface />
   497→      ) : (
   498→        <BetaInterface />
   499→      )}
   500→
   501→      {/* Accent ambient glow keyed to the current sector — HIDDEN in Beta mode */}
