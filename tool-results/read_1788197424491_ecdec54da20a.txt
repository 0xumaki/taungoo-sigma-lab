    13→function useLiveTimestamp() {
    14→  const [ts, setTs] = React.useState("T+04:17:22");
    15→  React.useEffect(() => {
    16→    let sec = 4 * 3600 + 17 * 60 + 22;
    17→    const id = setInterval(() => {
    18→      sec += 1;
    19→      const h = String(Math.floor(sec / 3600)).padStart(2, "0");
    20→      const m = String(Math.floor((sec % 3600) / 60)).padStart(2, "0");
    21→      const s = String(sec % 60).padStart(2, "0");
    22→      setTs(`T+${h}:${m}:${s}`);
    23→    }, 1000);
    24→    return () => clearInterval(id);
    25→  }, []);
    26→  return ts;
    27→}

    45→function useCountUp(target: number, duration = 1500, delay = 0) {
    46→  const [val, setVal] = React.useState(0);
    47→  React.useEffect(() => {
    48→    let raf: number;
    49→    const start = performance.now() + delay;
    50→    const tick = (now: number) => {
    51→      if (now < start) { raf = requestAnimationFrame(tick); return; }
    52→      const t = Math.min(1, (now - start) / duration);
    53→      const eased = 1 - Math.pow(1 - t, 3);
    54→      setVal(target * eased);
    55→      if (t < 1) raf = requestAnimationFrame(tick);
    56→    };
    57→    raf = requestAnimationFrame(tick);
    58→    return () => cancelAnimationFrame(raf);
    59→  }, [target, duration, delay]);
    60→  return val;
    61→}

    63→/**
    64→ * useTextScramble — cyberpunk decoder effect.
    65→ * Returns a string that starts as random chars and resolves to the target.
    66→ */
    67→const SCRAMBLE_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZΣ0123456789";
    68→function randomScramble(len: number) {
    69→  let s = "";
    70→  for (let i = 0; i < len; i++) s += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
    71→  return s;
    72→}
    73→function useTextScramble(target: string, duration = 1200, delay = 0) {
    74→  // Initialize with the TARGET string (deterministic) so SSR output matches the
    75→  // first client render — the scramble then starts inside useEffect (client-only),
    76→  // which fires immediately after hydration. Fixes the hydration mismatch warning.
    77→  const [text, setText] = React.useState(() => target);
    78→  React.useEffect(() => {
    79→    let raf: number;
    80→    const start = performance.now() + delay;
    81→    const tick = (now: number) => {
    82→      if (now < start) {
    83→        // Update scramble rapidly while waiting for delay to pass
    84→        setText(randomScramble(target.length));
    85→        raf = requestAnimationFrame(tick);
    86→        return;
    87→      }
    88→      const t = Math.min(1, (now - start) / duration);
    89→      // Number of chars resolved = ceil(t * length)
    90→      const resolved = Math.ceil(t * target.length);
    91→      let result = "";
    92→      for (let i = 0; i < target.length; i++) {
    93→        if (i < resolved || target[i] === " ") result += target[i];
    94→        else result += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
    95→      }
    96→      setText(result);
    97→      if (t < 1) raf = requestAnimationFrame(tick);
    98→    };
    99→    raf = requestAnimationFrame(tick);
   100→    return () => cancelAnimationFrame(raf);
   101→  }, [target, duration, delay]);
   102→  return text;
   103→}

   117→/**
   118→ * useMouseParallax — returns spring-smoothed x/y motion values (in px) for a
   119→ * subtle parallax effect based on mouse position relative to viewport center.
   120→ * strength: how many px to move at max (e.g. 20 = ±20px)
   121→ */
   122→function useMouseParallax(strength = 20) {
   123→  const x = useMotionValue(0);
   124→  const y = useMotionValue(0);
   125→  const sx = useSpring(x, { stiffness: 80, damping: 20, restDelta: 0.1 });
   126→  const sy = useSpring(y, { stiffness: 80, damping: 20, restDelta: 0.1 });
   127→  React.useEffect(() => {
   128→    if (typeof window === "undefined") return;
   129→    const finePointer = window.matchMedia("(pointer: fine)").matches;
   130→    if (!finePointer) return;
   131→    // Respect prefers-reduced-motion: skip mouse parallax entirely.
   132→    // (Hero already disables scroll parallax via useReducedMotion in render.)
   133→    const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   134→    if (reducedMotion) return;
   135→    const onMove = (e: MouseEvent) => {
   136→      const cx = window.innerWidth / 2;
   137→      const cy = window.innerHeight / 2;
   138→      // Normalize to [-1, 1]
   139→      const nx = (e.clientX - cx) / cx;
   140→      const ny = (e.clientY - cy) / cy;
   141→      x.set(nx * strength);
   142→      y.set(ny * strength);
   143→    };
   144→    window.addEventListener("mousemove", onMove);
   145→    return () => window.removeEventListener("mousemove", onMove);
   146→  }, [strength, x, y]);
   147→  return { x: sx, y: sy };
   148→}

   150→/**
   151→ * useMagnetic — returns a callback ref that attaches a magnetic-cursor effect.
   152→ * Apply to a button/anchor for subtle cursor attraction (max `strength` px pull).
   153→ * Respects prefers-reduced-motion (skips entirely) + touch devices (pointer: coarse).
   154→ * Uses a callback ref pattern so it works for both always-mounted elements AND
   155→ * conditionally-rendered ones (e.g., a multi-step form's submit button that only
   156→ * appears on the last step). Exported so other beta components (Contact submit)
   157→ * can reuse the same px-based magnetic feel as the Hero CTA.
   158→ */
   159→export function useMagnetic<T extends HTMLElement = HTMLAnchorElement>(strength = 6) {
   160→  const [el, setEl] = React.useState<T | null>(null);
   161→  const ref = React.useCallback((node: T | null) => setEl(node), []);
   162→  React.useEffect(() => {
   163→    if (!el) return;
   164→    if (typeof window === "undefined") return;
   165→    const finePointer = window.matchMedia("(pointer: fine)").matches;
   166→    if (!finePointer) return;
   167→    const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   168→    if (reducedMotion) return;
   169→    const onMove = (e: MouseEvent) => {
   170→      const r = el.getBoundingClientRect();
   171→      const cx = r.left + r.width / 2;
   172→      const cy = r.top + r.height / 2;
   173→      const dx = (e.clientX - cx) / r.width;
   174→      const dy = (e.clientY - cy) / r.height;
   175→      // Only attract when cursor is within 1.5x button size
   176→      if (Math.abs(dx) < 1.5 && Math.abs(dy) < 1.5) {
   177→        el.style.transform = `translate(${dx * strength}px, ${dy * strength}px)`;
   178→      } else {
   179→        el.style.transform = "translate(0, 0)";
   180→      }
   181→    };
   182→    const onLeave = () => { el.style.transform = "translate(0, 0)"; };
   183→    window.addEventListener("mousemove", onMove);
   184→    el.addEventListener("mouseleave", onLeave);
   185→    return () => {
   186→      window.removeEventListener("mousemove", onMove);
   187→      el.removeEventListener("mouseleave", onLeave);
   188→      el.style.transform = "";
   189→    };
   190→  }, [el, strength]);
   191→  return ref;
   192→}

   194→/**
   195→ * useTypingText — types out text char-by-char, then blinks cursor.
   196→ * Loops after a pause.
   197→ */
   198→function useTypingText(text: string, speed = 50, startDelay = 0, pauseAfter = 2500) {
   199→  const [output, setOutput] = React.useState("");
   200→  const [done, setDone] = React.useState(false);
   201→  React.useEffect(() => {
   202→    let i = 0;
   203→    let timer: ReturnType<typeof setTimeout>;
   204→    let pauseTimer: ReturnType<typeof setTimeout>;
   205→    const startTimer = setTimeout(function type() {
   206→      if (i <= text.length) {
   207→        setOutput(text.slice(0, i));
   208→        setDone(i === text.length);
   209→        i++;
   210→        timer = setTimeout(type, speed);
   211→      } else {
   212→        // Pause then restart
   213→        pauseTimer = setTimeout(() => {
   214→          i = 0;
   215→          setOutput("");
   216→          setDone(false);
   217→          timer = setTimeout(type, speed);
   218→        }, pauseAfter);
   219→      }
   220→    }, startDelay);
   221→    return () => {
   222→      clearTimeout(startTimer);
   223→      clearTimeout(timer);
   224→      clearTimeout(pauseTimer);
   225→    };
   226→  }, [text, speed, startDelay, pauseAfter]);
   227→  return { output, done };
   228→}

   230→/**
   231→ * useLiveBars — returns an array of heights (0-100) that update on an interval.
   232→ * Used for the throughput monitor visualization.
   233→ *
   234→ * HYDRATION SAFETY: the initial state is deterministic (all bars at 50) so the
   235→ * SSR HTML matches the first client render. The randomization only starts inside
   236→ * useEffect (client-only) — same pattern as useTextScramble + useCountUp above.
   237→ * Without this guard, Math.random() in the useState initializer would produce
   238→ * different bar values on server vs client → hydration mismatch warning.
   239→ */
   240→function useLiveBars(count = 12, intervalMs = 1200) {
   241→  const [bars, setBars] = React.useState<number[]>(() =>
   242→    Array.from({ length: count }, () => 50)
   243→  );
   244→  React.useEffect(() => {
   245→    // First tick: seed real random values (client-only).
   246→    setBars(Array.from({ length: count }, () => 20 + Math.random() * 70));
   247→    const id = setInterval(() => {
   248→      // Skip oscilloscope churn when the tab is hidden — bars are decorative.
   249→      if (document.hidden) return;
   250→      setBars((prev) => {
   251→        // Shift left, push new value at end (oscilloscope-style)
   252→        const next = [...prev.slice(1), 20 + Math.random() * 75];
   253→        return next;
   254→      });
   255→    }, intervalMs);
   256→    return () => clearInterval(id);
   257→  }, [count, intervalMs]);
   258→  return bars;
   259→}
