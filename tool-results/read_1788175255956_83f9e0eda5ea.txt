    13→function useLiveTimestamp() {
    14→  const [ts, setTs] = React.useState("T+04:17:22");
    15→  React.useEffect(() => {
    16→    let sec = 4 * 3600 + 17 * 60 + 22;
    17→    const id = setInterval(() => {
    18→      sec += 1;
    19→      const h = String(Math.floor(sec / 3600)).padStart(2, "0");
    20→      const m = String(Math.floor((sec % 3600) / 60)).padStart(2, "0");
    21→      const s = String(sec % 60).padStart(2, "0");
    22→      setTs(`T+${h}:${m}:${s}`);
    23→    }, 1000);
    24→    return () => clearInterval(id);
    25→  }, []);
    26→  return ts;
    27→}

    42→function useCountUp(target: number, duration = 1500, delay = 0) {
    43→  const [val, setVal] = React.useState(0);
    44→  React.useEffect(() => {
    45→    let raf: number;
    46→    const start = performance.now() + delay;
    47→    const tick = (now: number) => {
    48→      if (now < start) { raf = requestAnimationFrame(tick); return; }
    49→      const t = Math.min(1, (now - start) / duration);
    50→      const eased = 1 - Math.pow(1 - t, 3);
    51→      setVal(target * eased);
    52→      if (t < 1) raf = requestAnimationFrame(tick);
    53→    };
    54→    raf = requestAnimationFrame(tick);
    55→    return () => cancelAnimationFrame(raf);
    56→  }, [target, duration, delay]);
    57→  return val;
    58→}

    60→/**
    61→ * useTextScramble — cyberpunk decoder effect.
    62→ * Returns a string that starts as random chars and resolves to the target.
    63→ */
    64→const SCRAMBLE_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZΣ0123456789";
    65→function randomScramble(len: number) {
    66→  let s = "";
    67→  for (let i = 0; i < len; i++) s += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
    68→  return s;
    69→}
    70→function useTextScramble(target: string, duration = 1200, delay = 0) {
    71→  // Start with a scrambled string so user sees decode effect from t=0
    72→  const [text, setText] = React.useState(() => randomScramble(target.length));
    73→  React.useEffect(() => {
    74→    let raf: number;
    75→    const start = performance.now() + delay;
    76→    const tick = (now: number) => {
    77→      if (now < start) {
    78→        // Update scramble rapidly while waiting for delay to pass
    79→        setText(randomScramble(target.length));
    80→        raf = requestAnimationFrame(tick);
    81→        return;
    82→      }
    83→      const t = Math.min(1, (now - start) / duration);
    84→      // Number of chars resolved = ceil(t * length)
    85→      const resolved = Math.ceil(t * target.length);
    86→      let result = "";
    87→      for (let i = 0; i < target.length; i++) {
    88→        if (i < resolved || target[i] === " ") result += target[i];
    89→        else result += SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
    90→      }
    91→      setText(result);
    92→      if (t < 1) raf = requestAnimationFrame(tick);
    93→    };
    94→    raf = requestAnimationFrame(tick);
    95→    return () => cancelAnimationFrame(raf);
    96→  }, [target, duration, delay]);
    97→  return text;
    98→}

   112→/**
   113→ * useMouseParallax — returns spring-smoothed x/y motion values (in px) for a
   114→ * subtle parallax effect based on mouse position relative to viewport center.
   115→ * strength: how many px to move at max (e.g. 20 = ±20px)
   116→ */
   117→function useMouseParallax(strength = 20) {
   118→  const x = useMotionValue(0);
   119→  const y = useMotionValue(0);
   120→  const sx = useSpring(x, { stiffness: 80, damping: 20, restDelta: 0.1 });
   121→  const sy = useSpring(y, { stiffness: 80, damping: 20, restDelta: 0.1 });
   122→  React.useEffect(() => {
   123→    if (typeof window === "undefined") return;
   124→    const finePointer = window.matchMedia("(pointer: fine)").matches;
   125→    if (!finePointer) return;
   126→    const onMove = (e: MouseEvent) => {
   127→      const cx = window.innerWidth / 2;
   128→      const cy = window.innerHeight / 2;
   129→      // Normalize to [-1, 1]
   130→      const nx = (e.clientX - cx) / cx;
   131→      const ny = (e.clientY - cy) / cy;
   132→      x.set(nx * strength);
   133→      y.set(ny * strength);
   134→    };
   135→    window.addEventListener("mousemove", onMove);
   136→    return () => window.removeEventListener("mousemove", onMove);
   137→  }, [strength, x, y]);
   138→  return { x: sx, y: sy };
   139→}

   141→/**
   142→ * useMagnetic — returns a ref + mousemove handler for magnetic button effect.
   143→ * Apply to a button/anchor for subtle cursor attraction.
   144→ */
   145→function useMagnetic(strength = 6) {
   146→  const ref = React.useRef<HTMLAnchorElement>(null);
   147→  React.useEffect(() => {
   148→    const el = ref.current;
   149→    if (!el) return;
   150→    if (typeof window === "undefined") return;
   151→    const finePointer = window.matchMedia("(pointer: fine)").matches;
   152→    if (!finePointer) return;
   153→    const onMove = (e: MouseEvent) => {
   154→      const r = el.getBoundingClientRect();
   155→      const cx = r.left + r.width / 2;
   156→      const cy = r.top + r.height / 2;
   157→      const dx = (e.clientX - cx) / r.width;
   158→      const dy = (e.clientY - cy) / r.height;
   159→      // Only attract when cursor is within 1.5x button size
   160→      if (Math.abs(dx) < 1.5 && Math.abs(dy) < 1.5) {
   161→        el.style.transform = `translate(${dx * strength}px, ${dy * strength}px)`;
   162→      } else {
   163→        el.style.transform = "translate(0, 0)";
   164→      }
   165→    };
   166→    const onLeave = () => { el.style.transform = "translate(0, 0)"; };
   167→    window.addEventListener("mousemove", onMove);
   168→    el.addEventListener("mouseleave", onLeave);
   169→    return () => {
   170→      window.removeEventListener("mousemove", onMove);
   171→      el.removeEventListener("mouseleave", onLeave);
   172→    };
   173→  }, [strength]);
   174→  return ref;
   175→}

   177→/**
   178→ * useTypingText — types out text char-by-char, then blinks cursor.
   179→ * Loops after a pause.
   180→ */
   181→function useTypingText(text: string, speed = 50, startDelay = 0, pauseAfter = 2500) {
   182→  const [output, setOutput] = React.useState("");
   183→  const [done, setDone] = React.useState(false);
   184→  React.useEffect(() => {
   185→    let i = 0;
   186→    let timer: ReturnType<typeof setTimeout>;
   187→    let pauseTimer: ReturnType<typeof setTimeout>;
   188→    const startTimer = setTimeout(function type() {
   189→      if (i <= text.length) {
   190→        setOutput(text.slice(0, i));
   191→        setDone(i === text.length);
   192→        i++;
   193→        timer = setTimeout(type, speed);
   194→      } else {
   195→        // Pause then restart
   196→        pauseTimer = setTimeout(() => {
   197→          i = 0;
   198→          setOutput("");
   199→          setDone(false);
   200→          timer = setTimeout(type, speed);
   201→        }, pauseAfter);
   202→      }
   203→    }, startDelay);
   204→    return () => {
   205→      clearTimeout(startTimer);
   206→      clearTimeout(timer);
   207→      clearTimeout(pauseTimer);
   208→    };
   209→  }, [text, speed, startDelay, pauseAfter]);
   210→  return { output, done };
   211→}

   213→/**
   214→ * useLiveBars — returns an array of heights (0-100) that update on an interval.
   215→ * Used for the throughput monitor visualization.

   217→function useLiveBars(count = 12, intervalMs = 1200) {
   218→  const [bars, setBars] = React.useState<number[]>(() =>
   219→    Array.from({ length: count }, () => 20 + Math.random() * 70)
   220→  );
   221→  React.useEffect(() => {
   222→    const id = setInterval(() => {
   223→      setBars((prev) => {
   224→        // Shift left, push new value at end (oscilloscope-style)
   225→        const next = [...prev.slice(1), 20 + Math.random() * 75];
   226→        return next;
   227→      });
   228→    }, intervalMs);
   229→    return () => clearInterval(id);
   230→  }, [count, intervalMs]);
   231→  return bars;
   232→}
